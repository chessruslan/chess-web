import 'dart:async';
import 'package:supabase_flutter/supabase_flutter.dart' as supa;
import 'package:realtime_client/realtime_client.dart' as rt;

/// Простая обёртка над Supabase Realtime Broadcast для сигналинга WebRTC.
/// События: 'offer' / 'answer' / 'ice' / 'ready'
class SupabaseSignaling {
  SupabaseSignaling._();
  static final instance = SupabaseSignaling._();

  final supa.SupabaseClient _sb = supa.Supabase.instance.client;

  /// Храним созданные каналы, чтобы отписаться при завершении
  final Map<String, rt.RealtimeChannel> _channels = {};

  rt.RealtimeChannel _ensure(String roomId) {
    final key = roomId.trim();
    final existed = _channels[key];
    if (existed != null) return existed;

    final ch = _sb.channel('room:$key');
    ch.subscribe(); // без await ок
    _channels[key] = ch;
    return ch;
  }

  Future<void> leaveRoom(String roomId) async {
    final key = roomId.trim();
    final ch = _channels.remove(key);
    if (ch != null) {
      try {
        await ch.unsubscribe();
      } catch (_) {}
    }
  }

  // =========== OUT (отправка) ===========

  Future<void> sendOffer({
    required String roomId,
    required String sdp,
  }) async {
    final ch = _ensure(roomId);
    await ch.sendBroadcastMessage(event: 'offer', payload: {'sdp': sdp});
  }

  Future<void> sendAnswer({
    required String roomId,
    required String sdp,
  }) async {
    final ch = _ensure(roomId);
    await ch.sendBroadcastMessage(event: 'answer', payload: {'sdp': sdp});
  }

  /// Кандидата передаём как Map<String, dynamic>:
  /// {'candidate': '...', 'sdpMid': '...', 'sdpMLineIndex': 0}
  Future<void> sendIce({
    required String roomId,
    required Map<String, dynamic> candidate,
  }) async {
    final ch = _ensure(roomId);
    await ch.sendBroadcastMessage(event: 'ice', payload: candidate);
  }

  /// READY (рукопожатие) — любой участник может отправить: “я готов, пришли оффер ещё раз”
  Future<void> sendReady({required String roomId}) async {
    final ch = _ensure(roomId);
    await ch.sendBroadcastMessage(
      event: 'ready',
      payload: {'t': DateTime.now().toIso8601String()},
    );
  }

  // =========== ON (подписки) ===========
  // В НОВОМ API подписка делается через onBroadcastStream(event: ...)
  // и возвращается StreamSubscription. Это нужно менять и в voice_service.dart.

  /// Кто-то присоединился и просит переслать offer (READY)
  StreamSubscription onReady(
    String roomId,
    Future<void> Function() handler,
  ) {
    final ch = _ensure(roomId);
    return ch.onBroadcastStream(event: 'ready').listen((_) async {
      await handler();
    });
  }

  /// Подписка на offer (обычно сторона "join")
  StreamSubscription onOffer(
    String roomId,
    Future<void> Function(String sdp) handler,
  ) {
    final ch = _ensure(roomId);
    return ch.onBroadcastStream(event: 'offer').listen((payload) async {
      String sdp = _extractSdp(payload);
      if (sdp.isNotEmpty) await handler(sdp);
    });
  }

  /// Подписка на answer (обычно сторона "create")
  StreamSubscription onAnswer(
    String roomId,
    Future<void> Function(String sdp) handler,
  ) {
    final ch = _ensure(roomId);
    return ch.onBroadcastStream(event: 'answer').listen((payload) async {
      String sdp = _extractSdp(payload);
      if (sdp.isNotEmpty) await handler(sdp);
    });
  }

  /// Подписка на удалённые ICE-кандидаты (слушают обе стороны)
  StreamSubscription onRemoteIce(
    String roomId,
    Future<void> Function(Map<String, dynamic> cand) handler,
  ) {
    final ch = _ensure(roomId);
    return ch.onBroadcastStream(event: 'ice').listen((payload) async {
      // может прийти сразу {candidate:...} или внутри "payload"
      Map<String, dynamic> data = {};
      if (payload is Map && payload['payload'] is Map) {
        data = (payload['payload'] as Map).cast<String, dynamic>();
      } else if (payload is Map) {
        data = payload.cast<String, dynamic>();
      }
      await handler(data);
    });
  }

  // Вспомогательное: вытянуть SDP из разных форматов payload
  String _extractSdp(dynamic payload) {
    if (payload is Map && payload['sdp'] != null) {
      return payload['sdp'].toString();
    }
    if (payload is Map &&
        payload['payload'] is Map &&
        (payload['payload'] as Map)['sdp'] != null) {
      return (payload['payload'] as Map)['sdp'].toString();
    }
    return '';
  }
}
