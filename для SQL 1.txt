-- 1) Профили пользователей (связаны с auth.users)
create table if not exists public.profiles (
  id uuid primary key references auth.users(id) on delete cascade,
  nickname text unique not null,
  created_at timestamptz default now()
);

-- 2) Челленджи (вызовы на игру)
create table if not exists public.challenges (
  id uuid primary key default gen_random_uuid(),
  from_id uuid not null references public.profiles(id) on delete cascade,
  to_id   uuid not null references public.profiles(id) on delete cascade,
  color_pref text check (color_pref in ('white','black','random')) default 'random',
  status text check (status in ('pending','accepted','declined','cancelled')) default 'pending',
  match_id uuid null, -- сюда положим id матча после акцепта
  created_at timestamptz default now()
);

-- 3) Матчи (партии)
create table if not exists public.matches (
  id uuid primary key default gen_random_uuid(),
  white_id uuid references public.profiles(id) on delete set null,
  black_id uuid references public.profiles(id) on delete set null,
  status text check (status in ('active','finished')) default 'active',
  fen text default 'rnbqkbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1',
  turn char(1) default 'w',  -- 'w' or 'b'
  result text,
  created_at timestamptz default now()
);

-- 4) Ходы
create table if not exists public.moves (
  id bigserial primary key,
  match_id uuid not null references public.matches(id) on delete cascade,
  ply int not null,           -- № полухода (1..)
  san text not null,
  from_sq text not null,
  to_sq text not null,
  fen_after text not null,
  made_by uuid references public.profiles(id),
  created_at timestamptz default now()
);

-- RLS
alter table public.profiles   enable row level security;
alter table public.challenges enable row level security;
alter table public.matches    enable row level security;
alter table public.moves      enable row level security;

-- Профили: читает любой, вставляет/обновляет только владелец
create policy if not exists profiles_select on public.profiles
for select using (true);

create policy if not exists profiles_insert_self on public.profiles
for insert with check (auth.uid() = id);

create policy if not exists profiles_update_self on public.profiles
for update using (auth.uid() = id);

-- Челленджи:
-- читать может участник (from_id или to_id)
create policy if not exists challenges_select on public.challenges
for select using (auth.uid() = from_id or auth.uid() = to_id);

-- создавать может любой аутентифицированный
create policy if not exists challenges_insert on public.challenges
for insert with check (auth.uid() = from_id);

-- обновлять может отправитель или получатель
create policy if not exists challenges_update on public.challenges
for update using (auth.uid() = from_id or auth.uid() = to_id);

-- Матчи:
-- читать могут участники
create policy if not exists matches_select on public.matches
for select using (auth.uid() = white_id or auth.uid() = black_id);

-- создавать может любой аутентифицированный (при акцепте челленджа)
create policy if not exists matches_insert on public.matches
for insert with check (auth.uid() is not null);

-- обновлять могут участники
create policy if not exists matches_update on public.matches
for update using (auth.uid() = white_id or auth.uid() = black_id);

-- Ходы:
-- читать могут участники матча
create policy if not exists moves_select on public.moves
for select using (
  exists (select 1 from public.matches m
          where m.id = match_id
          and (auth.uid() = m.white_id or auth.uid() = m.black_id))
);

-- добавлять может участник матча
create policy if not exists moves_insert on public.moves
for insert with check (
  exists (select 1 from public.matches m
          where m.id = match_id
          and (auth.uid() = m.white_id or auth.uid() = m.black_id))
);

-- индексы (ускоряем подписки и выборки)
create index if not exists idx_challenges_to_id on public.challenges(to_id);
create index if not exists idx_moves_match on public.moves(match_id);
